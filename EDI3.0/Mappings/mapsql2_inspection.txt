PROCEDURE mlockmap
PARAMETERS llDisplay 
lcTrnType = PADR(THISFORMSET.Form1.cboTransactions.VALUE,3)
lcMapset  = PADR(MAPGRID.cMapSet,3)
lcVersion = PADR(THISFORMSET.Form1.cboVersion.Value,12)
lcUserID  = SYCEDIPD.CLockUser 
lcFolder  = ALLTRIM(THISFORMSET.Form1.txtLocalFolder.Value)
lcFolder   = ALLTRIM(lcFolder + IIF(RIGHT(lcFolder,1)<>"\","\"," "))
lcFileHdr  = ''
lcFileDtl  = ''
lcEDIType  = 'S'
lcHdrOrder = ''
lcDtlorder = ''
THISFORMSET.mGetFilesName(@lcFileHdr,@lcFileDtl,@lcEDIType,@lcHdrOrder,@lcDtlorder)
IF (!EMPTY(lcFileHdr) AND FILE(lcFolder+lcFileHdr+'.'+lcMapset)) OR ;
   (!EMPTY(lcFileDtl) AND FILE(lcFolder+lcFileDtl+'.'+lcMapset))
  lnAnsr=MESSAGEBOX("the file "+lcFolder+lcFileHdr+"."+lcMapset+" Already Exist, OverWrite",4+32, 'Error') 
  IF lnAnsr = 7
    RETURN
  ENDIF
ENDIF
IF lcEdiType = 'R' AND !llDisplay
   lcSqlStatmnt = "SELECT * FROM MapStats where cMapSet ='"+lcMapSEt+"' AND cVersion ='"+ ;
                  lcVersion+"' AND Type = 'R' AND lLocked = '1'"  
   lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt,"MapStats")
   IF lnRes = 1
     IF !EOF('MapStats')
      =MESSAGEBOX("There are transaction # "+MapStats.cTranType+" Locked ...Could not continue",0+16,'Error')
      RETURN
     ENDIF
   ENDIF
ENDIF
IF !EMPTY(lcFileHdr)
  IF lcEdiType $ 'LT'
     lcSqlStatmnt = "SELECT * FROM "+lcFileHdr+" where cVer ='"+lcMapSEt+"'" 
     IF lcEdiType = 'L'
      lcSqlStatmnt = lcSqlStatmnt && + "AND cPrint_Id ='"+lcVersion+"'"
     ENDIF
     lcSqlStatmnt = lcSqlStatmnt + " ORDER BY " + lcHdrOrder 
  ELSE
    lcSqlStatmnt = "SELECT * FROM "+lcFileHdr+" where cMapSet ='"+lcMapSEt+"' AND cVersion ='"+;
                      lcVersion+"' ORDER BY " + lcHdrOrder 
  ENDIF
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt,"lcFileHdr")
ENDIF
IF !EMPTY(lcFileDtl)
  lcSqlStatmnt = "SELECT * FROM "+lcFileDtl+" where cMapSet ='"+lcMapSEt+"' AND cVersion ='"+;
                 lcVersion+"' ORDER BY " + lcDtlorder 
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt,"lcFileDtl")
ENDIF
IF lcEdiType = 'L'
  lcSqlStatmnt = "SELECT * FROM SYCASNHD where cVer ='"+lcMapSEt+"'"
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt,"SYCASNHD")
  lcSqlStatmnt = "SELECT * FROM SYCASNDT where cVer ='"+lcMapSEt+"'"
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt,"SYCASNDT")
ENDIF

IF lnRes = 1
  lcSqlStatmnt = "UPDATE mapstats SET lockuser ='"+lcUserID+"',lLocked=1,lockdare =GETDATE() WHERE (ctrantype+cversion+cmapset+type='" + ;
                 lcTrnType +"'+'"+lcVersion+"'+'"+lcMapset+"'+'"+lcEDIType+"')"
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
  IF USED('lcFileHdr')
    SELECT lcFileHdr
    COPY TO lcFolder+lcFileHdr+"."+lcMapSet
  ENDIF
   IF USED('lcFileDtl')
    SELECT lcFileDtl
    COPY TO lcFolder+lcFileDtl+"."+lcMapSet
  ENDIF
  IF lcEdiType = 'L'
    IF USED('SYCASNHD')
      SELECT SYCASNHD
      COPY TO lcFolder+"SYCASNHD."+lcMapSet
    ENDIF
    IF USED('SYCASNDT')
      SELECT SYCASNDT
      COPY TO lcFolder+"SYCASNDT."+lcMapSet
    ENDIF
  ENDIF
  IF !llDisplay
    =MESSAGEBOX('Download Successfully          ',0+64, 'Download Mappings')
  ENDIF
  THISFORMSET.mgetmappings 
ENDIF
IF USED('MapStats')
  USE IN MapStats
ENDIF
IF USED('lcFileHdr')
  USE IN lcFileHdr
ENDIF
IF USED('lcFileDtl')
  USE IN lcFileDtl
ENDIF
IF USED('SYCASNHD')
  USE IN SYCASNHD
ENDIF
IF USED('SYCASNDT')
  USE IN SYCASNDT
ENDIF

ENDPROC
PROCEDURE muploadmap
lcTrnType = PADR(THISFORMSET.Form1.cboTransactions.VALUE,3)
lcMapset  = PADR(MAPGRID.cMapSet,3)
lcVersion = PADR(THISFORMSET.Form1.cboVersion.Value,12)
lcUserID  = MAPGRID.CLockUser 
lcFolder  = ALLTRIM(THISFORMSET.Form1.txtLocalFolder.Value)
lcFolder   = ALLTRIM(lcFolder + IIF(RIGHT(lcFolder,1)<>"\","\"," "))
lcFileHdr  = ''
lcFileDtl  = ''
lcEDIType  = 'S'
lcHdrOrder = ''
lcDtlorder = ''
THISFORMSET.mGetFilesName(@lcFileHdr,@lcFileDtl,@lcEDIType,@lcHdrOrder,@lcDtlorder)
IF !EMPTY(lcFileHdr) AND !FILE(lcFolder+lcFileHdr+'.'+lcMapset)
  MESSAGEBOX("File " + lcFolder+lcFileHdr+'.'+lcMapset + "Does Not Exist",0+16,'Error')
ENDIF
IF !EMPTY(lcFileDtl) AND !FILE(lcFolder+lcFileDtl+'.'+lcMapset)
  MESSAGEBOX("File " + lcFolder+lcFileHdr+'.'+lcMapset + "Does Not Exist",0+16,'Error')
ENDIF
lnAnser=MESSAGEBOX('Are you sur that you want to upload this map?',4+32, 'Upload Mappings')
IF lnAnser = 7
  RETURN
ENDIF
lnTotalRes = 1
lcSqlStatmnt = "Begin Transaction"
lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)

IF !EMPTY(lcFileHdr)
  IF lcEDIType  $ 'LT'
     lcSqlStatmnt = "DELETE  FROM "+lcFileHdr+" where cVer ='"+lcMapSEt+"'" 
     IF lcEdiType = 'L'
       lcSqlStatmnt = lcSqlStatmnt  && + "AND cPrint_Id ='"+lcVersion+"'"
     ENDIF
  ELSE
    lcSqlStatmnt = "DELETE  FROM "+lcFileHdr+" where cMapSet ='"+lcMapSEt+"' AND cVersion ='"+lcVersion+"'"
  ENDIF
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
  IF lcEdiType = 'L'
     lcSqlStatmnt = "DELETE  FROM SYCASNHD where cVer ='"+lcMapSEt+"'" 
     lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
     lcSqlStatmnt = "DELETE  FROM SYCASNDT where cVer ='"+lcMapSEt+"'" 
     lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
  ENDIF
ENDIF
lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
IF !EMPTY(lcFileDtl) AND lnTotalRes = 1
  lcSqlStatmnt = "DELETE  FROM "+lcFileDtl+" where cMapSet ='"+lcMapSEt+"' AND cVersion ='"+lcVersion+"'"
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
ENDIF
lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 

IF lnTotalRes = 1
  IF !EMPTY(lcFileHdr)
    IF !USED(lcFileHdr)
      USE  lcFolder+lcFileHdr+'.'+lcMapSet IN 0
    ENDIF
    SELECT (lcFileHdr)
    LOCATE
    SCAN REST WHILE lnRes = 1
      WAIT WINDOW  lcFileHdr + '  : ' + STR(RECNO()) NOWAIT
      SCATTER MEMVAR MEMO
      lcSqlStatmnt = THISFORMSET.mBuildString(lcFileHdr)
      lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
    ENDSCAN
    lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
  ENDIF
  IF !EMPTY(lcFileDtl)  AND lnTotalRes = 1
    IF !USED(lcFileDtl)
      USE  lcFolder+lcFileDtl+'.'+lcMapSet IN 0
    ENDIF
    SELECT (lcFileDtl)
    LOCATE
    SCAN REST WHILE lnRes = 1
      WAIT WINDOW  lcFileDtl + '  : ' + STR(RECNO()) NOWAIT
      SCATTER MEMVAR MEMO
      lcSqlStatmnt = THISFORMSET.mBuildString(lcFileDtl)
      lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
    ENDSCAN
    lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
  ENDIF
  IF lcEdiType = 'L' AND lnTotalRes = 1
    IF !USED('SYCASNHD')
      USE lcFolder+'SYCASNHD.'+lcMapSet IN 0
    ENDIF
    SELECT SYCASNHD
    LOCATE
    SCAN REST WHILE lnRes = 1
      WAIT WINDOW  'SYCASNHD  : ' + STR(RECNO()) NOWAIT
      SCATTER MEMVAR MEMO
      lcSqlStatmnt = THISFORMSET.mBuildString('SYCASNHD')
      lnRes = SQLEXEC(THIS.lnHand,lcSqlStatmnt)
    ENDSCAN
    lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
    IF lnTotalRes = 1
    IF !USED('SYCASNDT')
      USE  lcFolder+'SYCASNDT.'+lcMapSet IN 0
    ENDIF
    SELECT SYCASNDT
    LOCATE
    SCAN REST WHILE lnRes = 1
      WAIT WINDOW  'SYCASNDT : ' + STR(RECNO()) NOWAIT
      SCATTER MEMVAR MEMO
      lcSqlStatmnt = THISFORMSET.mBuildString('SYCASNDT')
      lnRes = SQLEXEC(THIS.lnHand,lcSqlStatmnt)
    ENDSCAN
    lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
    ENDIF
  ENDIF
  
  lcRelease = PADL(ALLTRIM(STR(VAL(MAPGRID.crelaseno)+1)),6,'0')
  lcSqlStatmnt = "UPDATE mapstats SET lockuser ='',lLocked=0,lockdare =GETDATE() ,cRelaseNo = '"+lcRelease+"' WHERE (ctrantype+cversion+cmapset+type='" + ;
                lcTrnType +"'+'"+lcVersion+"'+'"+lcMapset+"'+'"+lcEDIType+"')"
                
  lcSqlStatmnt_PD = "UPDATE SYCEDIPD SET DEDIT_DATE=GETDATE() WHERE (ctrantype+cversion+cmapset+type='" + ;
                    lcTrnType +"'+'"+lcVersion+"'+'"+lcMapset+"'+'"+lcEDIType+"')"

                
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
  lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
  
  *lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt_PD)
  *lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
  
  lcSqlStatmnt = "SELECT * FROM MapHstry WHERE cTranType = '000'"
  lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt,'MapHstry')
  lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
  SELECT MapHstry
  SCATTER MEMVAR BLANK
  
  m.cTrantype  = lcTrnType
  m.cVersion  = lcVersion
  m.cMapSet   = lcMapset
  m.cRelaseno = lcRelease
  m.BuildNo   = MAPGRID.BuildNo
  m.TpRevision= MAPGRID.TpRevision
  m.modifRef  = MAPGRID.modifRef
  m.cUserId   = lcUserID
 lcSqlStatmnt = THISFORMSET.mBuildString('MapHstry')
 lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
 lnTotalRes = IIF(lnRes = 1 ,lnTotalRes,lnRes) 
ENDIF
IF lnTotalRes = 1
   lcSqlStatmnt = "Commit"
   lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
   =MESSAGEBOX('Upload Successfully          ',0+64, 'UpLoad Mappings')
    THISFORMSET.mgetmappings 
ELSE 
   lcSqlStatmnt = "ROLLBACK"
   lnRes=SQLEXEC(THIS.lnHand,lcSqlStatmnt)
  MESSAGEBOX("Couldn't Delete the original Map Records",0+16,'Error')
  RETURN
ENDIF
IF USED(lcFileHdr)
  USE IN (lcFileHdr)
ENDIF
IF USED(lcFileDtl)
  USE IN (lcFileDtl)
ENDIF
IF USED('MapHstry')
  USE IN MapHstry
ENDIF
IF USED('SYCASNHD')
  USE IN SYCASNHD
ENDIF
IF USED('SYCASNDT')
  USE IN SYCASNDT
ENDIF

ENDPROC
PROCEDURE mgetfilesname
PARAMETERS lcFileHdr, lcFileDtl, lcEDIType , lcHdrOrder , lcDtlorder
DO CASE
  CASE  '204' $ lcTrnType 
    lcFileHdr = 'SYCEDISM'
    lcHdrOrder= 'cmapset+cversion+loop_id+STR(f_order,2)'
  CASE  '753' $ lcTrnType 
    lcFileHdr = 'SYCEDIRF'
    lcHdrOrder= 'cmapset+cversion+loop_id+STR(f_order,2)'
  CASE  '754' $ lcTrnType 
    lcEDIType     = 'R' 
    lcFileHdr  = 'SYCEDISG'
    lcFileDtl  = 'SYCEDISF'
    lcHdrOrder = 'cmapset+cversion+segid'
    lcDtlorder = 'cmapset+cversion+segid+STR(f_order,2)'
  CASE  '810' $ lcTrnType 
    IF SUBSTR(SYCEDIPD.CTRANTYPE,1,1)= 'R'  
      lcEDIType     = 'R' 
      lcFileHdr  = 'SYCEDISG'
      lcFileDtl  = 'SYCEDISF'
      lcHdrOrder = 'cmapset+cversion+segid'
      lcDtlorder = 'cmapset+cversion+segid+STR(f_order,2)'
    ELSE
      lcFileHdr = 'SYCEDIFS'
      lcHdrOrder= 'cmapset+cversion+loop_id+STR(f_order,2)'
    ENDIF          
  CASE  '811' $ lcTrnType 
    lcFileHdr = 'SYCEDIFS'
    lcHdrOrder= 'cmapset+cversion+loop_id+STR(f_order,2)'
  CASE  '812' $ lcTrnType 
    IF SUBSTR(SYCEDIPD.CTRANTYPE,1,1)= 'I'
      lcEDIType     = 'R' 
      lcFileHdr  = 'SYCEDISG'
      lcFileDtl  = 'SYCEDISF'
      lcHdrOrder = 'cmapset+cversion+segid'
      lcDtlorder = 'cmapset+cversion+segid+STR(f_order,2)'
    ELSE
      lcFileHdr = 'SYCEDICD'
      lcHdrOrder= 'cmapset+cversion+loop_id+STR(f_order,2)'
    ENDIF
  CASE  '816' $ lcTrnType 
    lcEDIType     = 'R' 
    lcFileHdr  = 'SYCEDISG'
    lcFileDtl  = 'SYCEDISF'
    lcHdrOrder = 'cmapset+cversion+segid'
    lcDtlorder = 'cmapset+cversion+segid+STR(f_order,2)'
  CASE  '820' $ lcTrnType 
    lcEDIType     = 'R' 
    lcFileHdr  = 'SYCEDISG'
    lcFileDtl  = 'SYCEDISF'
    lcHdrOrder = 'cmapset+cversion+segid'
    lcDtlorder = 'cmapset+cversion+segid+STR(f_order,2)'

PROCEDURE mbuildstring
PARAMETERS lcfileName
LOCAL lcStr
lcStr = ""    && Default return string by empty string.
lcStr = "INSERT INTO " +lcfileName 
STORE "" TO lcStr1,lcStr2
*--Loop on the cursor fields.
FOR lnI=1 TO FCOUNT()
  *--Read field in variable.
  lcCField = FIELD(lnI)
  lcCFldDType = TYPE(lcCField)
  lcStr1 = lcStr1 +"["+lcCField+"],"
  IF UPPER(lcCField) = "ROWGUID"
     lcStr2 = lcStr2 +"newid(),"
  ELSE
    lcStr2 = lcStr2 +"?m."+lcCField+","
  ENDIF
ENDFOR

*--Remove last comma,
lcStr1 = SUBSTR(lcStr1,1,LEN(lcStr1)-1)
lcStr2 = SUBSTR(lcStr2,1,LEN(lcStr2)-1)
lcStr  = lcStr + " ( "+lcStr1+" ) VALUES ( "+lcStr2+" )"
RETURN lcStr



ENDPROC
