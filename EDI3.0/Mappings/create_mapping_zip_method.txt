PROCEDURE CreateMappingZip
LPARAMETERS tcFolder, tcMapSet
LOCAL lcFolder, lcMapSet, lcZipFile, lcConfigFile, lcCommand, loShell, lnResult

lcFolder = ADDBS(ALLTRIM(tcFolder))
lcMapSet = ALLTRIM(tcMapSet)
IF EMPTY(lcFolder) OR EMPTY(lcMapSet) OR !DIRECTORY(lcFolder)
  RETURN 87
ENDIF

lcZipFile = lcFolder + lcMapSet + ".ZIP"
lcConfigFile = lcFolder + "mapping.config"
IF FILE(lcZipFile)
  ERASE (lcZipFile)
ENDIF
IF FILE(lcConfigFile)
  ERASE (lcConfigFile)
ENDIF
IF PEMSTATUS(THIS.Form1, "chkPublishSQL", 5) AND THIS.Form1.chkPublishSQL.Value
  =STRTOFILE("PUBLISH_SQL=TRUE" + CHR(13) + CHR(10), lcConfigFile, 0)
ENDIF

loShell = CREATEOBJECT("WScript.Shell")
lcCommand = "powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -Command " + CHR(34) + ;
  "$map='" + STRTRAN(lcMapSet, "'", "''") + "';" + ;
  "$files=@(Get-ChildItem -LiteralPath '" + STRTRAN(lcFolder, "'", "''") + ;
  "' | Where-Object {-not $_.PSIsContainer -and (" + ;
  "$_.Extension -ieq ('.'+$map) -or $_.Extension -ieq '.FPT' -or " + ;
  "$_.Name -ieq 'mapping.config' -or " + ;
  "(( $_.Extension -ieq '.FRX' -or $_.Extension -ieq '.FRT') -and " + ;
  "$_.BaseName -like ('*'+$map)))});" + ;
  "if($files.Count -eq 0){exit 2};" + ;
  "Compress-Archive -LiteralPath $files.FullName -DestinationPath '" + ;
  STRTRAN(lcZipFile, "'", "''") + "' -Force;" + ;
  "if(Test-Path -LiteralPath '" + STRTRAN(lcZipFile, "'", "''") + ;
  "'){$files | Remove-Item -Force}else{exit 3}" + CHR(34)
lnResult = loShell.Run(lcCommand, 0, .T.)

IF lnResult = 0 AND FILE(lcZipFile)
  RETURN 0
ENDIF
RETURN IIF(lnResult = 0, 1, lnResult)
ENDPROC
